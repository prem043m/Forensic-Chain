Step 3: Handle Blockchain Transaction

async function handleSignatureAndTransfer(transferData, signatureData, modal) { const padContainer = modal.querySelector('#signature-pad'); const pad = modal.signaturePad; pad.setStatus('Processing blockchain transaction...', 'info'); pad.setButtonsDisabled(true); try { // Call your backend endpoint to create blockchain transaction const response = await fetch('/api/evidence/transfer', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ evidence_id: transferData.id, to_custodian: transferData.to, reason: transferData.reason, signed_at: signatureData.timestamp, signature_hash: signatureData.signatureHash }) }); const result = await response.json(); if (result.tx_hash) { pad.setStatus( `✓ Transfer recorded! TX: ${result.tx_hash.substring(0, 16)}...`, 'success' ); setTimeout(() => modal.remove(), 2500); } else { pad.setStatus('Transfer failed: ' + result.error, 'error'); } } catch (error) { pad.setStatus('Error: ' + error.message, 'error'); } finally { pad.setButtonsDisabled(false); } }
